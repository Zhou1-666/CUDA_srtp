check_sha() {
    [[ -f "$1" && ! -L "$1" ]] || return 2
    local digest
    digest=$(sha256sum -- "$1" | awk '{print $1}')
    printf 'INPUT_SHA256=%s path=%s\n' "$digest" "$1"
    [[ "$digest" == "$2" ]]
}
check_launcher() {
    local launcher=$1 expected_resolved=$2 expected_sha=$3 resolved digest
    [[ -L "$launcher" ]] || return 2
    resolved=$(readlink -f -- "$launcher") || return 2
    [[ "$resolved" == "$expected_resolved" && -f "$resolved" && ! -L "$resolved" ]] || return 2
    digest=$(sha256sum -- "$resolved" | awk '{print $1}') || return 2
    printf 'LAUNCHER_PATH=%s\nLAUNCHER_RESOLVED=%s\n' "$launcher" "$resolved"
    printf 'INPUT_SHA256=%s path=%s\n' "$digest" "$resolved"
    [[ "$digest" == "$expected_sha" ]]
}
